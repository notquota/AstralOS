#include "pci.h"

namespace Astral::PCI {

static Device devices[MAX_DEVICES]{};

static uint32_t totalDevices = 0;
static bool ready = false;

static constexpr uint16_t CONFIG_ADDRESS = 0xCF8;
static constexpr uint16_t CONFIG_DATA = 0xCFC;

static void out32(uint16_t port, uint32_t value) {
    asm volatile(
        "outl %0, %w1"
        :
        : "a"(value), "Nd"(port)
        : "memory"
    );
}

static uint32_t in32(uint16_t port) {
    uint32_t value = 0;

    asm volatile(
        "inl %w1, %0"
        : "=a"(value)
        : "Nd"(port)
        : "memory"
    );

    return value;
}

static uint32_t config_address(
    uint8_t bus,
    uint8_t slot,
    uint8_t function,
    uint8_t offset
) {
    return 0x80000000u |
        (static_cast<uint32_t>(bus) << 16) |
        (static_cast<uint32_t>(slot & 0x1F) << 11) |
        (static_cast<uint32_t>(function & 7) << 8) |
        (offset & 0xFC);
}

uint32_t read_config(
    uint8_t bus,
    uint8_t slot,
    uint8_t function,
    uint8_t offset
) {
    out32(
        CONFIG_ADDRESS,
        config_address(bus, slot, function, offset)
    );

    return in32(CONFIG_DATA);
}

void write_config(
    uint8_t bus,
    uint8_t slot,
    uint8_t function,
    uint8_t offset,
    uint32_t value
) {
    out32(
        CONFIG_ADDRESS,
        config_address(bus, slot, function, offset)
    );

    out32(CONFIG_DATA, value);
}

static void inspect(
    uint8_t bus,
    uint8_t slot,
    uint8_t function
) {
    const uint32_t identity =
        read_config(bus, slot, function, 0x00);

    const uint16_t vendor =
        static_cast<uint16_t>(identity);

    if (vendor == 0xFFFF)
        return;

    if (totalDevices >= MAX_DEVICES)
        return;

    const uint32_t classInfo =
        read_config(bus, slot, function, 0x08);

    const uint32_t headerInfo =
        read_config(bus, slot, function, 0x0C);

    const uint32_t interrupts =
        read_config(bus, slot, function, 0x3C);

    Device& d = devices[totalDevices++];

    d.bus = bus;
    d.slot = slot;
    d.function = function;

    d.vendorId = vendor;
    d.deviceId =
        static_cast<uint16_t>(identity >> 16);

    d.classCode =
        static_cast<uint8_t>(classInfo >> 24);

    d.subclass =
        static_cast<uint8_t>(classInfo >> 16);

    d.progIF =
        static_cast<uint8_t>(classInfo >> 8);

    d.revision =
        static_cast<uint8_t>(classInfo);

    d.headerType =
        static_cast<uint8_t>(headerInfo >> 16);

    d.interruptLine =
        static_cast<uint8_t>(interrupts);

    d.interruptPin =
        static_cast<uint8_t>(interrupts >> 8);
}

bool initialize() {
    totalDevices = 0;
    ready = false;

    for (uint32_t bus = 0; bus < 256; ++bus) {
        for (uint32_t slot = 0; slot < 32; ++slot) {
            uint8_t b = static_cast<uint8_t>(bus);
            uint8_t s = static_cast<uint8_t>(slot);

            uint32_t id =
                read_config(b, s, 0, 0x00);

            if ((id & 0xFFFF) == 0xFFFF)
                continue;

            uint32_t header =
                read_config(b, s, 0, 0x0C);

            uint8_t headerType =
                static_cast<uint8_t>(header >> 16);

            uint32_t functionCount =
                (headerType & 0x80) ? 8 : 1;

            for (uint32_t function = 0;
                 function < functionCount;
                 ++function) {
                inspect(
                    b,
                    s,
                    static_cast<uint8_t>(function)
                );
            }
        }
    }

    ready = true;
    return true;
}

bool initialized() {
    return ready;
}

uint32_t device_count() {
    return totalDevices;
}

bool device_at(
    uint32_t index,
    Device& output
) {
    if (!ready || index >= totalDevices)
        return false;

    output = devices[index];
    return true;
}

bool find_class(
    uint8_t classCode,
    uint8_t subclass,
    Device& output
) {
    if (!ready)
        return false;

    for (uint32_t i = 0; i < totalDevices; ++i) {
        const Device& d = devices[i];

        if (d.classCode == classCode &&
            d.subclass == subclass) {
            output = d;
            return true;
        }
    }

    return false;
}

bool find_device(
    uint16_t vendorId,
    uint16_t deviceId,
    Device& output
) {
    if (!ready)
        return false;

    for (uint32_t i = 0; i < totalDevices; ++i) {
        const Device& d = devices[i];

        if (d.vendorId == vendorId &&
            d.deviceId == deviceId) {
            output = d;
            return true;
        }
    }

    return false;
}

bool read_bar(
    const Device& device,
    uint8_t barIndex,
    Bar& bar
) {
    bar = {BarType::Invalid, 0, false};

    // Type 0 PCI headers have six BARs.
    // Type 1 PCI bridge headers have two.
    const uint8_t headerType =
        device.headerType & 0x7F;

    uint8_t maxBars = 0;

    if (headerType == 0)
        maxBars = 6;
    else if (headerType == 1)
        maxBars = 2;
    else
        return false;

    if (barIndex >= maxBars)
        return false;

    const uint8_t offset =
        static_cast<uint8_t>(0x10 + barIndex * 4);

    const uint32_t low = read_config(
        device.bus,
        device.slot,
        device.function,
        offset
    );

    if (low == 0)
        return false;

    if (low & 1) {
        bar.type = BarType::IO;
        bar.address =
            static_cast<uint64_t>(low & ~0x3u);
        return true;
    }

    bar.prefetchable = (low & 0x8) != 0;

    const uint32_t memoryType =
        (low >> 1) & 0x3;

    if (memoryType == 0) {
        bar.type = BarType::Memory32;
        bar.address =
            static_cast<uint64_t>(low & ~0xFu);
        return true;
    }

    if (memoryType == 2 &&
        barIndex + 1 < maxBars) {
        const uint32_t high = read_config(
            device.bus,
            device.slot,
            device.function,
            static_cast<uint8_t>(offset + 4)
        );

        bar.type = BarType::Memory64;

        bar.address =
            (static_cast<uint64_t>(high) << 32) |
            static_cast<uint64_t>(low & ~0xFu);

        return true;
    }

    return false;
}

}

static constexpr uint16_t INTEL = 0x8086;
static constexpr uint16_t I82540EM = 0x100E;
static constexpr uint16_t I82545EM = 0x100F;

static constexpr uint32_t REG_CTRL   = 0x0000;
static constexpr uint32_t REG_STATUS = 0x0008;
static constexpr uint32_t REG_ICR    = 0x00C0;
static constexpr uint32_t REG_IMC    = 0x00D8;

static constexpr uint32_t REG_RCTL   = 0x0100;
static constexpr uint32_t REG_TCTL   = 0x0400;
static constexpr uint32_t REG_TIPG   = 0x0410;

static constexpr uint32_t REG_RDBAL  = 0x2800;
static constexpr uint32_t REG_RDBAH  = 0x2804;
static constexpr uint32_t REG_RDLEN  = 0x2808;
static constexpr uint32_t REG_RDH    = 0x2810;
static constexpr uint32_t REG_RDT    = 0x2818;

static constexpr uint32_t REG_TDBAL  = 0x3800;
static constexpr uint32_t REG_TDBAH  = 0x3804;
static constexpr uint32_t REG_TDLEN  = 0x3808;
static constexpr uint32_t REG_TDH    = 0x3810;
static constexpr uint32_t REG_TDT    = 0x3818;

static constexpr uint32_t REG_RAL0   = 0x5400;
static constexpr uint32_t REG_RAH0   = 0x5404;

static constexpr uint32_t CTRL_RST  = 1u << 26;
static constexpr uint32_t CTRL_SLU  = 1u << 6;
static constexpr uint32_t CTRL_ASDE = 1u << 5;

static constexpr uint32_t STATUS_LU = 1u << 1;

static constexpr uint32_t RCTL_EN    = 1u << 1;
static constexpr uint32_t RCTL_BAM   = 1u << 15;
static constexpr uint32_t RCTL_SECRC = 1u << 26;

static constexpr uint32_t TCTL_EN  = 1u << 1;
static constexpr uint32_t TCTL_PSP = 1u << 3;

static constexpr uint8_t RX_DD  = 1u << 0;
static constexpr uint8_t RX_EOP = 1u << 1;

static constexpr uint8_t TX_EOP = 1u << 0;
static constexpr uint8_t TX_RS  = 1u << 3;
static constexpr uint8_t TX_DD  = 1u << 0;

static constexpr uint32_t RX_COUNT = 32;
static constexpr uint32_t TX_COUNT = 16;

static constexpr uint32_t BUFFER_SIZE = 2048;
static constexpr uint32_t MAX_FRAME = 1514;

static constexpr uint32_t RESET_TIMEOUT = 2000000;
static constexpr uint32_t TX_TIMEOUT = 2000000;

// ================================================
// Legacy E1000 descriptor formats
// ================================================

struct RxDescriptor {
    uint64_t address;
    uint16_t length;
    uint16_t checksum;
    uint8_t status;
    uint8_t errors;
    uint16_t special;
} __attribute__((packed));

struct TxDescriptor {
    uint64_t address;
    uint16_t length;
    uint8_t checksumOffset;
    uint8_t command;
    uint8_t status;
    uint8_t checksumStart;
    uint16_t special;
} __attribute__((packed));

static_assert(sizeof(RxDescriptor) == 16);
static_assert(sizeof(TxDescriptor) == 16);

static uint32_t txIndex = 0;

static uint32_t read(uint32_t offset) {
    return MMIO::read32(registers, offset);
}

static void write(
    uint32_t offset,
    uint32_t value
) {
    MMIO::write32(registers, offset, value);
}

static void barrier() {
    // Compiler ordering fence.
    // The DMA backend must also provide
    // coherent mappings for this platform.
    asm volatile("" ::: "memory");
}

static bool supported(
    const PCI::Device& device
) {
    return device.vendorId == INTEL &&
        (device.deviceId == I82540EM ||
         device.deviceId == I82545EM);
}

static bool find_adapter(
    PCI::Device& output
) {
    for (uint32_t i = 0;
         i < PCI::device_count();
         ++i) {

        PCI::Device d{};

        if (!PCI::device_at(i, d))
            continue;

        if (supported(d)) {
            output = d;
            return true;
        }
    }

    return false;
}

static bool valid_mac(
    const MacAddress& mac
) {
    bool allZero = true;
    bool allFF = true;

    for (uint32_t i = 0; i < 6; ++i) {
        if (mac.bytes[i] != 0)
            allZero = false;

        if (mac.bytes[i] != 0xFF)
            allFF = false;
    }

    return !allZero &&
           !allFF &&
           !(mac.bytes[0] & 1);
}

static void extract_mac(
    uint32_t low,
    uint32_t high
) {
    currentMac.bytes[0] =
        static_cast<uint8_t>(low);

    currentMac.bytes[1] =
        static_cast<uint8_t>(low >> 8);

    currentMac.bytes[2] =
        static_cast<uint8_t>(low >> 16);

    currentMac.bytes[3] =
        static_cast<uint8_t>(low >> 24);

    currentMac.bytes[4] =
        static_cast<uint8_t>(high);

    currentMac.bytes[5] =
        static_cast<uint8_t>(high >> 8);

    macValid =
        (high & (1u << 31)) &&
        valid_mac(currentMac);
}

static void program_mac() {
    uint32_t low =
        static_cast<uint32_t>(currentMac.bytes[0]) |
        (static_cast<uint32_t>(currentMac.bytes[1]) << 8) |
        (static_cast<uint32_t>(currentMac.bytes[2]) << 16) |
        (static_cast<uint32_t>(currentMac.bytes[3]) << 24);

    uint32_t high =
        static_cast<uint32_t>(currentMac.bytes[4]) |
        (static_cast<uint32_t>(currentMac.bytes[5]) << 8) |
        (1u << 31);

    write(REG_RAL0, low);
    write(REG_RAH0, high);
}

// ================================================
// Adapter discovery
// ================================================

bool initialize() {
    detected = false;
    active = false;

    registers = nullptr;
    macValid = false;
    counters = {};

    if (!PCI::initialized() ||
        !MMIO::available())
        return false;

    if (!find_adapter(pciDevice))
        return false;

    PCI::Bar bar{};

    if (!PCI::read_bar(pciDevice, 0, bar))
        return false;

    if (bar.type != PCI::BarType::Memory32 &&
        bar.type != PCI::BarType::Memory64)
        return false;

    registers = MMIO::map(
        bar.address,
        0x6000
    );

    if (!registers)
        return false;

    // Enable PCI MMIO access.
    uint32_t command = PCI::read_config(
        pciDevice.bus,
        pciDevice.slot,
        pciDevice.function,
        0x04
    );

    PCI::write_config(
        pciDevice.bus,
        pciDevice.slot,
        pciDevice.function,
        0x04,
        command | 0x2u
    );

    uint32_t low = read(REG_RAL0);
    uint32_t high = read(REG_RAH0);

    extract_mac(low, high);

    detected = true;

    return true;
}

// ================================================
// DMA allocation
// ================================================

static bool allocate_rings() {
    if (!DMA::available())
        return false;

    if (!DMA::allocate(
        RX_COUNT * sizeof(RxDescriptor),
        128,
        rxRingMemory
    )) {
        return false;
    }

    if (!DMA::allocate(
        TX_COUNT * sizeof(TxDescriptor),
        128,
        txRingMemory
    )) {
        return false;
    }

    DMA::zero(rxRingMemory);
    DMA::zero(txRingMemory);

    rxRing = reinterpret_cast<
        volatile RxDescriptor*
    >(rxRingMemory.virtualAddress);

    txRing = reinterpret_cast<
        volatile TxDescriptor*
    >(txRingMemory.virtualAddress);

    for (uint32_t i = 0; i < RX_COUNT; ++i) {
        if (!DMA::allocate(
            BUFFER_SIZE,
            16,
            rxBuffers[i]
        )) {
            return false;
        }

        DMA::zero(rxBuffers[i]);

        rxRing[i].address =
            rxBuffers[i].physicalAddress;
    }

    for (uint32_t i = 0; i < TX_COUNT; ++i) {
        if (!DMA::allocate(
            BUFFER_SIZE,
            16,
            txBuffers[i]
        )) {
            return false;
        }

        DMA::zero(txBuffers[i]);

        txRing[i].address =
            txBuffers[i].physicalAddress;

        // Descriptor initially available.
        txRing[i].status = TX_DD;
    }

    rxIndex = 0;
    txIndex = 0;

    barrier();

    return true;
}

// ================================================
// NIC reset
// ================================================

static bool reset_adapter() {
    // Mask all device interrupts; this driver
    // currently uses polling.
    write(REG_IMC, 0xFFFFFFFFu);

    write(REG_RCTL, 0);
    write(REG_TCTL, 0);

    uint32_t control = read(REG_CTRL);

    write(REG_CTRL, control | CTRL_RST);

    bool finished = false;

    for (uint32_t i = 0;
         i < RESET_TIMEOUT;
         ++i) {

        if (!(read(REG_CTRL) & CTRL_RST)) {
            finished = true;
            break;
        }

        asm volatile("pause");
    }

    if (!finished)
        return false;

    write(REG_IMC, 0xFFFFFFFFu);

    // Clear pending causes.
    (void)read(REG_ICR);

    // Ask the controller to establish link.
    write(
        REG_CTRL,
        read(REG_CTRL) | CTRL_SLU | CTRL_ASDE
    );

    if (macValid)
        program_mac();

    return true;
}

// ================================================
// Receive and transmit rings
// ================================================

static void configure_receive() {
    const uint64_t physical =
        rxRingMemory.physicalAddress;

    write(
        REG_RDBAL,
        static_cast<uint32_t>(physical)
    );

    write(
        REG_RDBAH,
        static_cast<uint32_t>(physical >> 32)
    );

    write(
        REG_RDLEN,
        RX_COUNT * sizeof(RxDescriptor)
    );

    write(REG_RDH, 0);
    write(REG_RDT, RX_COUNT - 1);

    // 2048-byte receive buffers.
    // Accept broadcast frames and strip FCS.
    write(
        REG_RCTL,
        RCTL_EN | RCTL_BAM | RCTL_SECRC
    );
}

static void configure_transmit() {
    const uint64_t physical =
        txRingMemory.physicalAddress;

    write(
        REG_TDBAL,
        static_cast<uint32_t>(physical)
    );

    write(
        REG_TDBAH,
        static_cast<uint32_t>(physical >> 32)
    );

    write(
        REG_TDLEN,
        TX_COUNT * sizeof(TxDescriptor)
    );

    write(REG_TDH, 0);
    write(REG_TDT, 0);

    // IPG configuration for copper Ethernet.
    write(REG_TIPG, 0x0060200A);

    write(
        REG_TCTL,
        TCTL_EN |
        TCTL_PSP |
        (0x10u << 4) |
        (0x40u << 12)
    );
}

bool start() {
    if (active)
        return true;

    if (!detected || !registers ||
        !macValid)
        return false;

    if (!allocate_rings())
        return false;

    if (!reset_adapter())
        return false;

    // Enable bus mastering only after
    // the DMA descriptors are prepared.
    uint32_t command = PCI::read_config(
        pciDevice.bus,
        pciDevice.slot,
        pciDevice.function,
        0x04
    );

    PCI::write_config(
        pciDevice.bus,
        pciDevice.slot,
        pciDevice.function,
        0x04,
        command | 0x6u
    );

    barrier();

    configure_receive();
    configure_transmit();

    active = true;

    return true;
}

    if (!detected || !registers)

    if (!detected || !macValid)

// ================================================
// Ethernet transmission
// ================================================

bool send(
    const uint8_t* frame,
    uint16_t length
) {
    if (!active || !frame ||
        length < 14 ||
        length > MAX_FRAME)
        return false;

    const uint32_t slot = txIndex;

    volatile TxDescriptor& descriptor =
        txRing[slot];

    // Synchronous polling means we don't
    // reuse the descriptor until completion.
    if (!(descriptor.status & TX_DD))
        return false;

    uint8_t* destination =
        static_cast<uint8_t*>(
            txBuffers[slot].virtualAddress
        );

    for (uint32_t i = 0; i < length; ++i)
        destination[i] = frame[i];

    descriptor.length = length;
    descriptor.checksumOffset = 0;
    descriptor.checksumStart = 0;
    descriptor.special = 0;

    // EOP + request descriptor status writeback.
    descriptor.command = TX_EOP | TX_RS;
    descriptor.status = 0;

    barrier();

    txIndex = (txIndex + 1) % TX_COUNT;

    write(REG_TDT, txIndex);

    // Poll for completion.
    for (uint32_t i = 0;
         i < TX_TIMEOUT;
         ++i) {

        if (descriptor.status & TX_DD) {
            ++counters.transmitted;
            return true;
        }

        asm volatile("pause");
    }

    ++counters.txErrors;
    return false;
}

// ================================================
// Ethernet reception
// ================================================

uint32_t poll(uint32_t budget) {
    if (!active || budget == 0)
        return 0;

    uint32_t processed = 0;

    while (processed < budget) {
        const uint32_t slot = rxIndex;

        volatile RxDescriptor& descriptor =
            rxRing[slot];

        if (!(descriptor.status & RX_DD))
            break;

        barrier();

        const uint16_t length =
            descriptor.length;

        const bool complete =
            (descriptor.status & RX_EOP) != 0;

        const bool error =
            descriptor.errors != 0;

        if (complete && !error &&
            length >= 14 &&
            length <= MAX_FRAME) {

            if (receiveCallback) {
                const uint8_t* data =
                    static_cast<const uint8_t*>(
                        rxBuffers[slot].virtualAddress
                    );

                receiveCallback(data, length);

                ++counters.received;
            } else {
                ++counters.dropped;
            }
        } else {
            // Multi-descriptor and oversized packets
            // aren't supported in this early driver.
            ++counters.dropped;
        }

        descriptor.status = 0;
        descriptor.errors = 0;
        descriptor.length = 0;

        barrier();

        rxIndex = (rxIndex + 1) % RX_COUNT;

        // Return descriptor ownership to NIC.
        write(REG_RDT, slot);

        ++processed;
    }

    return processed;
}

#include "wifi.h"

#include "../../hardware/pci.h"

#include <stdint.h>

namespace Astral::WiFi {

// ---------------------------------------------
// Limits
// ---------------------------------------------

static constexpr uint32_t MAX_ADAPTERS = 16;
static constexpr uint32_t MAX_NETWORKS = 32;

// ---------------------------------------------
// PCI vendor identifiers
// ---------------------------------------------

static constexpr uint16_t VENDOR_INTEL = 0x8086;
static constexpr uint16_t VENDOR_ATHEROS = 0x168C;
static constexpr uint16_t VENDOR_BROADCOM = 0x14E4;
static constexpr uint16_t VENDOR_REALTEK = 0x10EC;
static constexpr uint16_t VENDOR_MEDIATEK = 0x14C3;
static constexpr uint16_t VENDOR_RALINK = 0x1814;

// ---------------------------------------------
// Driver state
// ---------------------------------------------

static Status current{};

static Adapter adapters[MAX_ADAPTERS]{};
static Network networks[MAX_NETWORKS]{};

static uint32_t adapterCount = 0;
static uint32_t networkCount = 0;

static bool ready = false;

// ---------------------------------------------
// Vendor recognition
// ---------------------------------------------

static Vendor identify_vendor(uint16_t id) {
    switch (id) {
        case VENDOR_INTEL:
            return Vendor::Intel;

        case VENDOR_ATHEROS:
            return Vendor::QualcommAtheros;

        case VENDOR_BROADCOM:
            return Vendor::Broadcom;

        case VENDOR_REALTEK:
            return Vendor::Realtek;

        case VENDOR_MEDIATEK:
        case VENDOR_RALINK:
            return Vendor::MediaTek;

        default:
            return Vendor::Unknown;
    }
}

const char* vendor_name(Vendor vendor) {
    switch (vendor) {
        case Vendor::Intel:
            return "Intel";

        case Vendor::QualcommAtheros:
            return "Qualcomm Atheros";

        case Vendor::Broadcom:
            return "Broadcom";

        case Vendor::Realtek:
            return "Realtek";

        case Vendor::MediaTek:
            return "MediaTek";

        default:
            return "Unknown";
    }
}

// ---------------------------------------------
// PCI classification
// ---------------------------------------------

static bool is_candidate(
    const PCI::Device& device
) {
    if (device.classCode != 0x02)
        return false;

    // PCI subclass 0x80 is "Other Network
    // Controller", not exclusively Wi-Fi.
    //
    // A wireless chipset still needs to be
    // confirmed using vendor/device IDs.
    return device.subclass == 0x80;
}

// ---------------------------------------------
// State management
// ---------------------------------------------

static void clear_state() {
    current = {};
    current.state = State::Unavailable;

    adapterCount = 0;
    networkCount = 0;

    for (uint32_t i = 0; i < MAX_ADAPTERS; ++i)
        adapters[i] = {};

    for (uint32_t i = 0; i < MAX_NETWORKS; ++i)
        networks[i] = {};
}

// ---------------------------------------------
// PCI discovery
// ---------------------------------------------

bool initialize() {
    clear_state();
    ready = false;

    if (!PCI::initialized())
        return false;

    for (uint32_t i = 0;
         i < PCI::device_count();
         ++i) {

        PCI::Device device{};

        if (!PCI::device_at(i, device))
            continue;

        if (!is_candidate(device))
            continue;

        if (adapterCount >= MAX_ADAPTERS)
            break;

        Adapter& adapter =
            adapters[adapterCount++];

        adapter.present = true;

        adapter.bus = device.bus;
        adapter.slot = device.slot;
        adapter.function = device.function;

        adapter.vendorId = device.vendorId;
        adapter.deviceId = device.deviceId;

        adapter.vendor =
            identify_vendor(device.vendorId);

        // No hardware-specific driver has
        // been implemented or registered.
        adapter.driverLoaded = false;
        adapter.firmwareLoaded = false;
    }

    current.detectedAdapters = adapterCount;

    if (adapterCount == 0) {
        current.state = State::Unavailable;
        ready = true;
        return true;
    }

    // The first candidate is selected for
    // display in the Settings interface.
    current.adapter = adapters[0];

    // All current wireless candidates need
    // a chipset-specific driver.
    current.state = State::DriverRequired;

    current.radioEnabled = false;
    current.scanSupported = false;
    current.connectionSupported = false;

    current.discoveredNetworks = 0;

    ready = true;
    return true;
}

bool initialized() {
    return ready;
}

const Status& status() {
    return current;
}

uint32_t adapter_count() {
    return adapterCount;
}

bool adapter_at(
    uint32_t index,
    Adapter& output
) {
    if (!ready || index >= adapterCount)
        return false;

    output = adapters[index];
    return true;
}

// ---------------------------------------------
// Discovered Wi-Fi networks
// ---------------------------------------------

uint32_t network_count() {
    return networkCount;
}

bool network_at(
    uint32_t index,
    Network& output
) {
    if (!ready || index >= networkCount)
        return false;

    output = networks[index];
    return true;
}

// ---------------------------------------------
// Wireless operations
// ---------------------------------------------

bool scan() {
    if (!ready ||
        !current.adapter.present ||
        !current.scanSupported)
        return false;

    // No radio backend is installed.
    return false;
}

bool connect(
    const char* ssid,
    const char* passphrase
) {
    if (!ssid || !*ssid)
        return false;

    (void)passphrase;

    if (!ready ||
        !current.adapter.present ||
        !current.connectionSupported)
        return false;

    // Association, encryption and firmware
    // interfaces are not yet implemented.
    return false;
}

void disconnect() {
    networkCount = 0;

    for (uint32_t i = 0; i < MAX_NETWORKS; ++i)
        networks[i] = {};

    current.discoveredNetworks = 0;

    if (!ready)
        return;

    current.state = current.adapter.present
        ? State::DriverRequired
        : State::Unavailable;
}

#pragma once

#include <stdint.h>

namespace Astral::Network {

struct MacAddress {
    uint8_t bytes[6];
};

struct IPv4Address {
    uint8_t bytes[4];
};

inline uint16_t read16(const uint8_t* p) {
    return static_cast<uint16_t>(
        (static_cast<uint16_t>(p[0]) << 8) |
        p[1]
    );
}

inline void write16(uint8_t* p, uint16_t value) {
    p[0] = static_cast<uint8_t>(value >> 8);
    p[1] = static_cast<uint8_t>(value);
}

inline void copy(
    uint8_t* dst,
    const uint8_t* src,
    uint32_t length
) {
    for (uint32_t i = 0; i < length; ++i)
        dst[i] = src[i];
}

inline bool equal(
    const uint8_t* a,
    const uint8_t* b,
    uint32_t length
) {
    for (uint32_t i = 0; i < length; ++i) {
        if (a[i] != b[i])
            return false;
    }
    return true;
}

inline bool same_ip(
    const IPv4Address& a,
    const IPv4Address& b
) {
    return equal(a.bytes, b.bytes, 4);
}

inline bool zero_ip(const IPv4Address& ip) {
    return ip.bytes[0] == 0 &&
           ip.bytes[1] == 0 &&
           ip.bytes[2] == 0 &&
           ip.bytes[3] == 0;
}

inline uint16_t checksum(
    const uint8_t* data,
    uint32_t length
) {
    uint32_t sum = 0;

    for (uint32_t i = 0; i < length; i += 2) {
        uint16_t word =
            static_cast<uint16_t>(data[i]) << 8;

        if (i + 1 < length)
            word |= data[i + 1];

        sum += word;
    }

    while (sum >> 16)
        sum = (sum & 0xFFFFu) + (sum >> 16);

    return static_cast<uint16_t>(~sum);
}

}

#include "ethernet.h"
#include "arp.h"
#include "ipv4.h"

#include "../drivers/network/e1000.h"

namespace Astral::Ethernet {

static bool ready = false;

static Network::MacAddress localMac{};
static Network::IPv4Address localIP{};

bool initialize(Network::IPv4Address ip) {
    ready = false;

    if (!E1000::running())
        return false;

    E1000::MacAddress hardware{};

    if (!E1000::mac_address(hardware))
        return false;

    Network::copy(localMac.bytes, hardware.bytes, 6);
    localIP = ip;

    ARP::initialize();

    ready = true;
    E1000::set_receive_callback(receive);

    return true;
}

bool initialized() {
    return ready;
}

Network::MacAddress local_mac() {
    return localMac;
}

Network::IPv4Address local_ip() {
    return localIP;
}

void set_local_ip(Network::IPv4Address ip) {
    localIP = ip;

    // Flush old link-layer address mappings when
    // the interface changes its IPv4 address.
    if (ready)
        ARP::initialize();
}

bool send(
    const Network::MacAddress& destination,
    uint16_t etherType,
    const uint8_t* payload,
    uint16_t length
) {
    if (!ready || !E1000::running())
        return false;

    if (length > MAX_PAYLOAD)
        return false;

    if (length != 0 && !payload)
        return false;

    uint8_t frame[HEADER_SIZE + MAX_PAYLOAD]{};

    Network::copy(frame, destination.bytes, 6);
    Network::copy(frame + 6, localMac.bytes, 6);

    Network::write16(frame + 12, etherType);

    if (length != 0)
        Network::copy(frame + HEADER_SIZE, payload, length);

    return E1000::send(
        frame,
        static_cast<uint16_t>(HEADER_SIZE + length)
    );
}

void receive(
    const uint8_t* frame,
    uint16_t length
) {
    if (!ready || !frame || length < HEADER_SIZE)
        return;

    static constexpr uint8_t broadcast[6] = {
        255, 255, 255, 255, 255, 255
    };

    if (!Network::equal(frame, localMac.bytes, 6) &&
        !Network::equal(frame, broadcast, 6))
        return;

    const uint16_t type = Network::read16(frame + 12);

    const uint8_t* payload = frame + HEADER_SIZE;
    const uint16_t payloadLength =
        static_cast<uint16_t>(length - HEADER_SIZE);

    if (type == TYPE_ARP) {
        ARP::receive(payload, payloadLength);
        return;
    }

    if (type == TYPE_IPV4) {
        Network::MacAddress source{};

        Network::copy(source.bytes, frame + 6, 6);

        IPv4::receive(
            payload,
            payloadLength,
            source
        );
    }
}

}

char history[HISTORY_CAPACITY][URL_CAPACITY]{};

void copy_string(
    char* destination,
    const char* source,
    uint32_t capacity
) {
    if (!destination || !capacity)
        return;

    uint32_t i = 0;

    if (source) {
        while (source[i] && i + 1 < capacity) {
            destination[i] = source[i];
            ++i;
        }
    }

    destination[i] = '\0';
}

uint32_t length(const char* text) {
    uint32_t i = 0;

    while (text && text[i] && i < URL_CAPACITY)
        ++i;

    return i;
}

bool inside(
    const Windows::Rect& r,
    int32_t x,
    int32_t y
) {
    return Windows::contains(r, x, y);
}

void button(
    const Windows::Rect& r,
    const char* label
) {
    Image::rectangle(
        r.x, r.y,
        r.width, r.height,
        0x304360
    );

    Image::text(
        r.x + 9,
        r.y + 13,
        label,
        WHITE
    );
}

void add_history(const char* url) {
    if (historyCount &&
        historyIndex + 1 < historyCount) {
        historyCount = historyIndex + 1;
    }

    if (historyCount >= HISTORY_CAPACITY) {
        for (uint32_t i = 1;
             i < HISTORY_CAPACITY;
             ++i) {
            copy_string(
                history[i - 1],
                history[i],
                URL_CAPACITY
            );
        }

        historyCount = HISTORY_CAPACITY - 1;
    }

    copy_string(
        history[historyCount],
        url,
        URL_CAPACITY
    );

    historyIndex = historyCount;
    ++historyCount;
}

bool visit(
    const char* url,
    bool saveHistory
) {
    if (!url || !url[0])
        return false;

    copy_string(address, url, URL_CAPACITY);

    if (!activePlugin.navigate) {
        state = PageState::PluginMissing;
        return false;
    }

    if (!activePlugin.online ||
        !activePlugin.online()) {
        state = PageState::Offline;
        return false;
    }

    state = PageState::Loading;

    if (!activePlugin.navigate(address)) {
        state = PageState::Error;
        return false;
    }

    if (saveHistory)
        add_history(address);

    state = PageState::Loaded;
    return true;
}

bool handle_key(char key) {
    if (!addressFocused)
        return false;

    uint32_t n = length(address);

    if (key == '\b') {
        if (n)
            address[n - 1] = '\0';

        return true;
    }

    if (key == '\n' || key == '\r') {
        addressFocused = false;

        // The Enter key has requested the internet.
        // The network stack hasn't RSVP'd.
        navigate(address);
        return true;
    }

    if (key >= 32 && key <= 126 &&
        n + 1 < URL_CAPACITY) {

        address[n] = key;
        address[n + 1] = '\0';
        return true;
    }

    return false;
}
